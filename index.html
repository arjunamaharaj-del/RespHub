        function initApp() {

        // Generate the tab sliders at the top

        function renderTabs() {

            const tabsContainer = document.getElementById('tabsContainer');

            tabsContainer.innerHTML = '';

            

            Object.keys(repoData).forEach(tabName => {

                const button = document.createElement('button');

                button.className = `px-4 py-2 rounded-lg text-xs font-medium tracking-tight whitespace-nowrap transition-all duration-200 ${

                    currentTab === tabName 

                    ? 'bg-[#1d1d1f] text-white font-semibold' 

                    : 'text-[#6e6e73] hover:text-[#1d1d1f] bg-transparent'

                }`;

                button.textContent = tabName;

                button.onclick = () => {

                    document.getElementById('globalSearch').value = '';

                    document.getElementById('searchBanner').classList.add('hidden');

                    currentTab = tabName;

                    renderTabs();

                    renderContent();

                };

                tabsContainer.appendChild(button);

            });

        }

        // Render current state details

        function renderContent() {

            const data = repoData[currentTab];

            

            const rctList = document.getElementById('rctList');

            const reviewList = document.getElementById('reviewList');

            const guidelineList = document.getElementById('guidelineList');


            document.getElementById('rctCount').textContent = data.rct.length;

            document.getElementById('reviewCount').textContent = data.review.length;

            document.getElementById('guidelineCount').textContent = data.guideline.length;


            rctList.innerHTML = data.rct.length ? data.rct.map(t => createCard(t, 'bg-blue-500')).join('') : emptyState();

            reviewList.innerHTML = data.review.length ? data.review.map(t => createCard(t, 'bg-purple-500')).join('') : emptyState();

            guidelineList.innerHTML = data.guideline.length ? data.guideline.map(t => createCard(t, 'bg-emerald-500')).join('') : emptyState();

            

            lucide.createIcons();

        }


        function emptyState() {

            return `<div class="text-xs text-[#86868b] py-6 text-center border border-dashed border-[#e8e8ed] rounded-xl font-light">No records captured in this track</div>`;

        }


        // Instant matching filter function across entire dataset

        function setupSearch() {

            const searchInput = document.getElementById('globalSearch');

            searchInput.addEventListener('input', (e) => {

                const query = e.target.value.toLowerCase().trim();

                if (!query) {

                    document.getElementById('searchBanner').classList.add('hidden');

                    renderContent();

                    return;

                }


                let matches = { rct: [], review: [], guideline: [] };

                let count = 0;


                Object.keys(repoData).forEach(tab => {

                    ['rct', 'review', 'guideline'].forEach(type => {

                        repoData[tab][type].forEach(study => {

                            if (study.title.toLowerCase().includes(query) || tab.toLowerCase().includes(query)) {

                                matches[type].push({

                                    title: `${study.title} (${tab})`,

                                    url: study.url

                                });

                                count++;

                            }

                        });

                    });

                });


                const banner = document.getElementById('searchBanner');

                const countText = document.getElementById('searchResultCount');

                banner.classList.remove('hidden');

                countText.textContent = `Found ${count} records across archive matching your query`;


                document.getElementById('rctCount').textContent = matches.rct.length;

                document.getElementById('reviewCount').textContent = matches.review.length;

                document.getElementById('guidelineCount').textContent = matches.guideline.length;


                document.getElementById('rctList').innerHTML = matches.rct.length ? matches.rct.map(t => createCard(t, 'bg-blue-500')).join('') : emptyState();

                document.getElementById('reviewList').innerHTML = matches.review.length ? matches.review.map(t => createCard(t, 'bg-purple-500')).join('') : emptyState();

                document.getElementById('guidelineList').innerHTML = matches.guideline.length ? matches.guideline.map(t => createCard(t, 'bg-emerald-500')).join('') : emptyState();

                

                lucide.createIcons();

            });

        }


        function clearSearch() {

            document.getElementById('globalSearch').value = '';

            document.getElementById('searchBanner').classList.add('hidden');

            renderContent();

        }


        function resetToDashboard() {

            currentTab = Object.keys(repoData)[0];

            clearSearch();

            renderTabs();

        }
